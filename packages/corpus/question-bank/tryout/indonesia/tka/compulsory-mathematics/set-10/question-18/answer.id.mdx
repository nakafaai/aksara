<BlockMath math="\begin{aligned}
72\text{ km per jam}&=\frac{72{.}000\text{ m}}{3{.}600\text{ detik}}=20\text{ m per detik} \\
t&=\frac{250+110}{20}=\frac{360}{20}=18\text{ detik}
\end{aligned}" />
