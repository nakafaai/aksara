<BlockMath math="\begin{aligned}
S&=50(75.6)=3{,}780 \\
S_A&=20(72)=1{,}440 \\
S_B&=3{,}780-1{,}440=2{,}340
\end{aligned}" />

<BlockMath math="\bar x_B=\frac{2{,}340}{30}=78" />
