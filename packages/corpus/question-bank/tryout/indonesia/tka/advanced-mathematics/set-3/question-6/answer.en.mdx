<BlockMath math="P(1)=4(1)-1=3,\qquad P(3)=7(3)-7=14" />

<BlockMath math="\begin{aligned}
a+b&=3 \\
3a+b&=14 \\
2a&=11\ \Rightarrow\ a=\frac{11}{2},\quad b=3-\frac{11}{2}=-\frac52
\end{aligned}" />
