<BlockMath math="4{.}500{.}000+450{.}000=4{.}950{.}000" />

<BlockMath math="5\%\times4{.}950{.}000=247{.}500" />

<BlockMath math="4{.}950{.}000-247{.}500=4{.}702{.}500" />
