<BlockMath math="\begin{aligned}
\text{total interest} &= 4{,}800{,}000\cdot1.5\%\cdot12=864{,}000\\
\text{instalment} &= \frac{4{,}800{,}000+864{,}000}{12}=\frac{5{,}664{,}000}{12}=472{,}000
\end{aligned}" />
